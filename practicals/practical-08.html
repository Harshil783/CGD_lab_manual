Implementation of 3D Transformations (Translation, Rotation, Scaling, Reflection and Shearing)

Aim

To implement 3D transformations: translation, rotation, scaling, reflection and shearing.

Input

Enter translation (tx ty tz): 50 30 20 Enter scaling (sx sy sz): 1.5 1.5 1.5 Enter rotation angle in degrees: 45 Enter shearing factors (shx shy): 0.5 0.5

Program / Code

#include <graphics.h>
#include <conio.h>
#include <math.h>
#include <stdio.h>

#define PI 3.14159265

int cx, cy;

// Cube (side 60) away from the origin so that every transformation is visible
float cube[8][3] = {
   {40, 20, 10}, {100, 20, 10}, {100, 80, 10}, {40, 80, 10},
   {40, 20, 70}, {100, 20, 70}, {100, 80, 70}, {40, 80, 70}
};
float t[8][3];

// Oblique (cabinet) projection of a 3D point to the screen
void project(float x, float y, float z, int *sx, int *sy) {
   *sx = cx + (int)(x - 0.35 * z);
   *sy = cy - (int)(y - 0.35 * z);
}

void drawCube(float p[][3], int color) {
  int i, j, x1, y1, x2, y2;
  setcolor(color);
  for (i = 0; i < 8; i++) {
    if (i < 4) j = (i + 1) % 4;
    else j = 4 + (i + 1) % 4;
    project(p[i][0], p[i][1], p[i][2], &x1, &y1);
    project(p[j][0], p[j][1], p[j][2], &x2, &y2);
    line(x1, y1, x2, y2);
    if (i < 4) {
        project(p[i + 4][0], p[i + 4][1], p[i + 4][2], &x2, &y2);
        line(x1, y1, x2, y2);
    }
  }
}



void drawAxes() {
  int x1, y1, x2, y2;
  setcolor(DARKGRAY);
  line(0, cy, getmaxx(), cy);              // X-axis
  line(cx, 0, cx, getmaxy());              // Y-axis
  project(0, 0, -250, &x1, &y1);
  project(0, 0, 250, &x2, &y2);
  line(x1, y1, x2, y2);                 // Z-axis
}

// type: 0 translate, 1 scale, 2 rotate X, 3 rotate Y, 4 rotate Z,
//      5 reflect XY plane, 6 reflect YZ plane, 7 reflect XZ plane, 8 shear
void transform(int type, float a, float b, float c) {
   int i;
   float x, y, z, r = a * PI / 180.0;
   for (i = 0; i < 8; i++) {
      x = t[i][0]; y = t[i][1]; z = t[i][2];
      switch (type) {
         case 0: x += a; y += b; z += c; break;
         case 1: x *= a; y *= b; z *= c; break;
         case 2: t[i][1] = y * cos(r) - z * sin(r);
              t[i][2] = y * sin(r) + z * cos(r);
               continue;
         case 3: t[i][0] = x * cos(r) + z * sin(r);
              t[i][2] = -x * sin(r) + z * cos(r);
               continue;
         case 4: t[i][0] = x * cos(r) - y * sin(r);
              t[i][1] = x * sin(r) + y * cos(r);
               continue;
         case 5: z = -z; break;
         case 6: x = -x; break;
         case 7: y = -y; break;
         case 8: x += a * z; y += b * z; break;
      }
      t[i][0] = x; t[i][1] = y; t[i][2] = z;
   }
}

void show(char *title) {
  cleardevice();
  drawAxes();
  drawCube(cube, WHITE);
  drawCube(t, LIGHTRED);
  setcolor(YELLOW);
  outtextxy(10, 10, title);
  outtextxy(10, 25, "White: Original Red: Transformed");
  getch();
}

        t[i][j] = cube[i][j];

int main() {
  int gd = DETECT, gm;
  float tx, ty, tz, sx, sy, sz, angle, shx, shy;

    printf("Enter translation (tx ty tz): ");
    scanf("%f%f%f", &tx, &ty, &tz);
    printf("Enter scaling (sx sy sz): ");
    scanf("%f%f%f", &sx, &sy, &sz);
    printf("Enter rotation angle in degrees: ");
    scanf("%f", &angle);
    printf("Enter shearing factors (shx shy): ");
    scanf("%f%f", &shx, &shy);

    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
    cx = getmaxx() / 2;
    cy = getmaxy() / 2;

    reset(); transform(0, tx, ty, tz); show("3D Translation");
    reset(); transform(1, sx, sy, sz); show("3D Scaling");
    reset(); transform(2, angle, 0, 0); show("3D Rotation about X-axis");
    reset(); transform(3, angle, 0, 0); show("3D Rotation about Y-axis");
    reset(); transform(4, angle, 0, 0); show("3D Rotation about Z-axis");
    reset(); transform(5, 0, 0, 0); show("3D Reflection about XY plane");
    reset(); transform(6, 0, 0, 0); show("3D Reflection about YZ plane");
    reset(); transform(7, 0, 0, 0); show("3D Reflection about XZ plane");
    reset(); transform(8, shx, shy, 0); show("3D Shearing");

    closegraph();
    return 0;
}
