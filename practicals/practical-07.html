Cohen-Sutherland Line Clipping Algorithm

Aim

To implement the Cohen-Sutherland line clipping algorithm.

Input

Enter clipping window (xmin ymin xmax ymax): 200 120 440 320 Enter line end points (x1 y1 x2 y2): 120 150 500 330

Program / Code

#include <graphics.h>
#include <conio.h>
#include <stdio.h>

#define INSIDE 0 // 0000
#define LEFT 1 // 0001
#define RIGHT 2 // 0010
#define BOTTOM 4 // 0100
#define TOP 8 // 1000

int xmin, ymin, xmax, ymax;

int computeCode(float x, float y) {
  int code = INSIDE;
  if (x < xmin) code |= LEFT;
  else if (x > xmax) code |= RIGHT;
  if (y < ymin) code |= TOP;      // screen y increases downwards
  else if (y > ymax) code |= BOTTOM;
  return code;
}

// Returns 1 if (part of) the line is visible; end points are updated
int cohenSutherland(float *x1, float *y1, float *x2, float *y2) {
   int code1 = computeCode(*x1, *y1);
   int code2 = computeCode(*x2, *y2);
   int codeOut;
   float x, y;

  while (1) {
   if ((code1 | code2) == 0) return 1; // trivially accepted
   if (code1 & code2) return 0;       // trivially rejected

    codeOut = code1 ? code1 : code2;
    if (codeOut & TOP) {
       x = *x1 + (*x2 - *x1) * (ymin - *y1) / (*y2 - *y1);
       y = ymin;
    } else if (codeOut & BOTTOM) {
       x = *x1 + (*x2 - *x1) * (ymax - *y1) / (*y2 - *y1);
       y = ymax;
    } else if (codeOut & RIGHT) {



           y = *y1 + (*y2 - *y1) * (xmax - *x1) / (*x2 - *x1);
           x = xmax;
        } else {
           y = *y1 + (*y2 - *y1) * (xmin - *x1) / (*x2 - *x1);
           x = xmin;
        }

        if (codeOut == code1) {
           *x1 = x; *y1 = y;
           code1 = computeCode(*x1, *y1);
        } else {
           *x2 = x; *y2 = y;
           code2 = computeCode(*x2, *y2);
        }
    }
}

int main() {
  int gd = DETECT, gm;
  float x1, y1, x2, y2;

    printf("Enter clipping window (xmin ymin xmax ymax): ");
    scanf("%d%d%d%d", &xmin, &ymin, &xmax, &ymax);
    printf("Enter line end points (x1 y1 x2 y2): ");
    scanf("%f%f%f%f", &x1, &y1, &x2, &y2);

    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    // Clipping window and original line (dotted)
    setcolor(CYAN);
    rectangle(xmin, ymin, xmax, ymax);
    setcolor(WHITE);
    setlinestyle(DOTTED_LINE, 0, 1);
    line((int) x1, (int) y1, (int) x2, (int) y2);
    outtextxy(10, 10, "Dotted: original line Red: clipped line");

    // Clipped line
    setlinestyle(SOLID_LINE, 0, THICK_WIDTH);
    if (cohenSutherland(&x1, &y1, &x2, &y2)) {
       setcolor(RED);
       line((int) x1, (int) y1, (int) x2, (int) y2);
    } else {
       setlinestyle(SOLID_LINE, 0, 1);
       outtextxy(10, 30, "Line is completely outside the window");
    }

    getch();
    closegraph();
    return 0;
}

Output

The clipping window is drawn, the original line is shown dotted and the clipped part of the line inside the window is shown in red.
