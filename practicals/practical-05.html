Flood Fill and Boundary Fill Algorithms

a) Implement flood fill algorithm.b) Implement boundary fill algorithm.

01a) Implement flood fill algorithm.

Aim

To implement the Flood Fill algorithm.

Input

No user input is required. All values are defined inside the program.

Program / Code

#include <graphics.h>
#include <conio.h>

#define MAXS 6500

int stkX[MAXS], stkY[MAXS], top = -1;

void push(int x, int y) {
  if (top < MAXS - 1) {
     top++;
     stkX[top] = x;
     stkY[top] = y;
  }
}

// Flood fill: replaces every connected pixel of oldColor with newColor
void floodFill(int x, int y, int newColor, int oldColor) {
   int cx, cy;
   if (oldColor == newColor || getpixel(x, y) != oldColor) return;

    putpixel(x, y, newColor);
    push(x, y);
    while (top >= 0) {
      cx = stkX[top];
      cy = stkY[top];
      top--;
      if (getpixel(cx + 1, cy) == oldColor) { putpixel(cx + 1, cy, newColor); push(cx + 1, cy); }
      if (getpixel(cx - 1, cy) == oldColor) { putpixel(cx - 1, cy, newColor); push(cx - 1, cy); }
      if (getpixel(cx, cy + 1) == oldColor) { putpixel(cx, cy + 1, newColor); push(cx, cy + 1); }
      if (getpixel(cx, cy - 1) == oldColor) { putpixel(cx, cy - 1, newColor); push(cx, cy - 1); }
    }
}

int main() {
  int gd = DETECT, gm;
  initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    setcolor(WHITE);
    rectangle(200, 150, 280, 230);
    floodFill(240, 190, GREEN, BLACK); // interior colour is BLACK



    getch();
    closegraph();
    return 0;
}

Output

A rectangle is drawn and its interior is filled with green colour.

02b) Implement boundary fill algorithm.

Aim

To implement the Boundary Fill algorithm.

Input

No user input is required. All values are defined inside the program.

Program / Code

#include <graphics.h>
#include <conio.h>

#define MAXS 6500

int stkX[MAXS], stkY[MAXS], top = -1;

void push(int x, int y) {
  if (top < MAXS - 1) {
     top++;
     stkX[top] = x;
     stkY[top] = y;
  }
}

// Boundary fill: fills until the boundary colour is reached
void boundaryFill(int x, int y, int fillColor, int boundaryColor) {



    int cx, cy, c;
    c = getpixel(x, y);
    if (c == boundaryColor || c == fillColor) return;

    putpixel(x, y, fillColor);
    push(x, y);
    while (top >= 0) {
      cx = stkX[top];
      cy = stkY[top];
      top--;

        c = getpixel(cx + 1, cy);
        if (c != boundaryColor && c != fillColor) { putpixel(cx + 1, cy, fillColor); push(cx + 1, cy); }
        c = getpixel(cx - 1, cy);
        if (c != boundaryColor && c != fillColor) { putpixel(cx - 1, cy, fillColor); push(cx - 1, cy); }
        c = getpixel(cx, cy + 1);
        if (c != boundaryColor && c != fillColor) { putpixel(cx, cy + 1, fillColor); push(cx, cy + 1); }
        c = getpixel(cx, cy - 1);
        if (c != boundaryColor && c != fillColor) { putpixel(cx, cy - 1, fillColor); push(cx, cy - 1); }
    }
}

int main() {
  int gd = DETECT, gm;
  initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    setcolor(WHITE);
    circle(250, 200, 40);
    boundaryFill(250, 200, RED, WHITE);

    getch();
    closegraph();
    return 0;
}

Output

A circle is drawn and its interior is filled with red colour up to the boundary.
