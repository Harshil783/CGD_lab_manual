Texture Mapping on Different Shapes using OpenGL

Aim

To apply texture mapping on different shapes (cube, sphere and cylinder) using OpenGL.

Input

No user input or image file is required. The checkerboard texture is generated inside the program.

Program / Code

#include <GL/glut.h>

#define TEX 64

GLubyte checker[TEX][TEX][3];
GLUquadric *quad;
float angle = 0.0f;

// Create a 64 x 64 red/white checkerboard texture in memory
void makeTexture() {
   int i, j;
   for (i = 0; i < TEX; i++) {
     for (j = 0; j < TEX; j++) {
         if (((i & 8) == 0) ^ ((j & 8) == 0)) {
             checker[i][j][0] = 255; checker[i][j][1] = 255; checker[i][j][2] = 255;
         } else {
             checker[i][j][0] = 200; checker[i][j][1] = 30; checker[i][j][2] = 30;
         }
     }
   }
}

void initTexture() {
  GLuint id;
  makeTexture();
  glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
  glGenTextures(1, &id);
  glBindTexture(GL_TEXTURE_2D, id);
  glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
  glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
  glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
  glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
  glTexEnvf(GL_TEXTURE_ENV, GL_TEXTURE_ENV_MODE, GL_REPLACE);
  glTexImage2D(GL_TEXTURE_2D, 0, GL_RGB, TEX, TEX, 0, GL_RGB,
          GL_UNSIGNED_BYTE, checker);
}

void texturedCube() {
  glBegin(GL_QUADS);
    // Front
    glTexCoord2f(0, 0); glVertex3f(-1, -1, 1);



      glTexCoord2f(1, 0); glVertex3f( 1, -1, 1);
      glTexCoord2f(1, 1); glVertex3f( 1, 1, 1);
      glTexCoord2f(0, 1); glVertex3f(-1, 1, 1);
      // Back
      glTexCoord2f(0, 0); glVertex3f( 1, -1, -1);
      glTexCoord2f(1, 0); glVertex3f(-1, -1, -1);
      glTexCoord2f(1, 1); glVertex3f(-1, 1, -1);
      glTexCoord2f(0, 1); glVertex3f( 1, 1, -1);
      // Top
      glTexCoord2f(0, 0); glVertex3f(-1, 1, 1);
      glTexCoord2f(1, 0); glVertex3f( 1, 1, 1);
      glTexCoord2f(1, 1); glVertex3f( 1, 1, -1);
      glTexCoord2f(0, 1); glVertex3f(-1, 1, -1);
      // Bottom
      glTexCoord2f(0, 0); glVertex3f(-1, -1, -1);
      glTexCoord2f(1, 0); glVertex3f( 1, -1, -1);
      glTexCoord2f(1, 1); glVertex3f( 1, -1, 1);
      glTexCoord2f(0, 1); glVertex3f(-1, -1, 1);
      // Right
      glTexCoord2f(0, 0); glVertex3f( 1, -1, 1);
      glTexCoord2f(1, 0); glVertex3f( 1, -1, -1);
      glTexCoord2f(1, 1); glVertex3f( 1, 1, -1);
      glTexCoord2f(0, 1); glVertex3f( 1, 1, 1);
      // Left
      glTexCoord2f(0, 0); glVertex3f(-1, -1, -1);
      glTexCoord2f(1, 0); glVertex3f(-1, -1, 1);
      glTexCoord2f(1, 1); glVertex3f(-1, 1, 1);
      glTexCoord2f(0, 1); glVertex3f(-1, 1, -1);
    glEnd();
}

void display() {
  glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
  glEnable(GL_TEXTURE_2D);

    // Textured cube
    glLoadIdentity();
    glTranslatef(-3.0f, 0.0f, -9.0f);
    glRotatef(angle, 1.0f, 1.0f, 0.0f);
    texturedCube();

    // Textured sphere
    glLoadIdentity();
    glTranslatef(0.0f, 0.0f, -9.0f);
    glRotatef(-90.0f, 1.0f, 0.0f, 0.0f);
    glRotatef(angle, 0.0f, 0.0f, 1.0f);
    gluSphere(quad, 1.3, 32, 32);

    // Textured cylinder
    glLoadIdentity();
    glTranslatef(3.0f, 0.0f, -9.0f);
    glRotatef(-60.0f, 1.0f, 0.0f, 0.0f);
    glRotatef(angle, 0.0f, 0.0f, 1.0f);



    glTranslatef(0.0f, 0.0f, -1.0f);
    gluCylinder(quad, 0.8, 0.8, 2.0, 32, 8);

    glDisable(GL_TEXTURE_2D);
    glutSwapBuffers();
}

void reshape(int w, int h) {
  if (h == 0) h = 1;
  glViewport(0, 0, w, h);
  glMatrixMode(GL_PROJECTION);
  glLoadIdentity();
  gluPerspective(45.0, (double) w / h, 1.0, 100.0);
  glMatrixMode(GL_MODELVIEW);
}

void update(int value) {
  angle += 1.0f;
  if (angle > 360.0f) angle -= 360.0f;
  glutPostRedisplay();
  glutTimerFunc(16, update, 0);
}

int main(int argc, char** argv) {
  glutInit(&argc, argv);
  glutInitDisplayMode(GLUT_DOUBLE | GLUT_RGB | GLUT_DEPTH);
  glutInitWindowSize(700, 450);
  glutCreateWindow("Texture Mapping on Different Shapes");

    glEnable(GL_DEPTH_TEST);
    glClearColor(0.1f, 0.1f, 0.15f, 1.0f);
    quad = gluNewQuadric();
    gluQuadricTexture(quad, GL_TRUE);
    initTexture();

    glutDisplayFunc(display);
    glutReshapeFunc(reshape);
    glutTimerFunc(16, update, 0);
    glutMainLoop();
    return 0;
}

Output

A cube, a sphere and a cylinder, each mapped with a red and white checkerboard texture, rotate in the OpenGL window.
