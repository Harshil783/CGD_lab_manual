01a) Implementation of Digital Differential Analyser algorithm

Aim

To implement the Digital Differential Analyser (DDA) line drawing algorithm.

Input

Enter x1 and y1: 100 100 Enter x2 and y2: 300 200

Program / Code

#include <graphics.h>
#include <conio.h>
#include <stdio.h>
#include <stdlib.h>

void DDA(int x1, int y1, int x2, int y2) {
  int dx = x2 - x1, dy = y2 - y1, i, steps;
  float x = x1, y = y1, xinc, yinc;

    steps = abs(dx) > abs(dy) ? abs(dx) : abs(dy);
    if (steps == 0) {
       putpixel(x1, y1, WHITE);
       return;
    }
    xinc = dx / (float) steps;
    yinc = dy / (float) steps;

    for (i = 0; i <= steps; i++) {
      putpixel((int)(x + 0.5), (int)(y + 0.5), WHITE);
      x += xinc;
      y += yinc;
    }
}

int main() {
  int gd = DETECT, gm, x1, y1, x2, y2;

    printf("Enter x1 and y1: ");
    scanf("%d%d", &x1, &y1);
    printf("Enter x2 and y2: ");
    scanf("%d%d", &x2, &y2);

    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
    DDA(x1, y1, x2, y2);

    getch();
    closegraph();
    return 0;
}

Output

A straight line is drawn from (100, 100) to (300, 200) in the graphics window.

02b) Implementation of Bresenham’s line drawing algorithm

Aim

To implement Bresenham’s line drawing algorithm.

Input

Enter x1 and y1: 100 100 Enter x2 and y2: 300 200

Program / Code

#include <graphics.h>
#include <conio.h>
#include <stdio.h>
#include <stdlib.h>

void bresenhamLine(int x1, int y1, int x2, int y2) {
  int dx = abs(x2 - x1), dy = abs(y2 - y1);
  int sx = (x1 < x2) ? 1 : -1;
  int sy = (y1 < y2) ? 1 : -1;
  int err = dx - dy, e2;

    while (1) {
      putpixel(x1, y1, WHITE);
      if (x1 == x2 && y1 == y2) break;
      e2 = 2 * err;
      if (e2 > -dy) { err -= dy; x1 += sx; }
      if (e2 < dx) { err += dx; y1 += sy; }
    }
}

int main() {
  int gd = DETECT, gm, x1, y1, x2, y2;




    printf("Enter x1 and y1: ");
    scanf("%d%d", &x1, &y1);
    printf("Enter x2 and y2: ");
    scanf("%d%d", &x2, &y2);

    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
    bresenhamLine(x1, y1, x2, y2);

    getch();
    closegraph();
    return 0;
}

Output

A straight line is drawn from (100, 100) to (300, 200) in the graphics window.

03c) Develop a program to display different types of lines.

Aim

To develop a program that displays different types of lines (solid, dotted, centre, dashed, user defined and thick).

Input

No user input is required. All values are defined inside the program.

Program / Code

#include <graphics.h>
#include <conio.h>

int main() {
  int gd = DETECT, gm;
  initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    setlinestyle(SOLID_LINE, 0, NORM_WIDTH);
    outtextxy(50, 70, "SOLID_LINE");



    line(50, 90, 300, 90);

    setlinestyle(DOTTED_LINE, 0, NORM_WIDTH);
    outtextxy(50, 120, "DOTTED_LINE");
    line(50, 140, 300, 140);

    setlinestyle(CENTER_LINE, 0, NORM_WIDTH);
    outtextxy(50, 170, "CENTER_LINE");
    line(50, 190, 300, 190);

    setlinestyle(DASHED_LINE, 0, NORM_WIDTH);
    outtextxy(50, 220, "DASHED_LINE");
    line(50, 240, 300, 240);

    setlinestyle(USERBIT_LINE, 0xF0F0, NORM_WIDTH);
    outtextxy(50, 270, "USERBIT_LINE (user defined pattern)");
    line(50, 290, 300, 290);

    setlinestyle(SOLID_LINE, 0, THICK_WIDTH);
    outtextxy(50, 320, "SOLID_LINE with THICK_WIDTH");
    line(50, 340, 300, 340);

    getch();
    closegraph();
    return 0;
}

Output

Six horizontal lines are displayed, each drawn with a different line style and labelled.
