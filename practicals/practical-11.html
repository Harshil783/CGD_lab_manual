Demonstrate Clipping on Different Shapes using OpenGL

Aim

To demonstrate clipping on different shapes (circle and triangle) using OpenGL clip planes.

Input

No user input is required. All values are defined inside the program.

Program / Code

#include <GL/glut.h>
#include <math.h>

// Draws a circle and a triangle; colours depend on whether clipping is on
void drawShapes(int clipped) {
   int i;

    if (clipped) glColor3f(1.0f, 0.0f, 0.0f);
    else       glColor3f(0.35f, 0.35f, 0.35f);
    glBegin(GL_TRIANGLE_FAN);
       glVertex2f(0.2f, 0.1f);
       for (i = 0; i <= 360; i += 10)
          glVertex2f(0.2f + 0.7f * (float) cos(i * 3.14159 / 180.0),
                 0.1f + 0.7f * (float) sin(i * 3.14159 / 180.0));
    glEnd();

    if (clipped) glColor3f(0.0f, 1.0f, 0.0f);
    else      glColor3f(0.55f, 0.55f, 0.55f);
    glBegin(GL_TRIANGLES);
       glVertex2f(-0.9f, -0.6f);
       glVertex2f( 0.9f, -0.4f);
       glVertex2f( 0.0f, 0.9f);
    glEnd();
}

void display() {
  // Clip plane equations: A*x + B*y + C*z + D >= 0 is kept
  GLdouble left[] = { 1.0, 0.0, 0.0, 0.5}; // x >= -0.5
  GLdouble right[] = {-1.0, 0.0, 0.0, 0.5}; // x <= 0.5
  GLdouble bottom[] = { 0.0, 1.0, 0.0, 0.5}; // y >= -0.5
  GLdouble top[] = { 0.0, -1.0, 0.0, 0.5}; // y <= 0.5

    glClear(GL_COLOR_BUFFER_BIT);
    glLoadIdentity();

    // 1. Shapes without clipping (grey)
    drawShapes(0);

    // 2. Same shapes clipped to the window (coloured)
    glClipPlane(GL_CLIP_PLANE0, left);
    glClipPlane(GL_CLIP_PLANE1, right);



    glClipPlane(GL_CLIP_PLANE2, bottom);
    glClipPlane(GL_CLIP_PLANE3, top);
    glEnable(GL_CLIP_PLANE0);
    glEnable(GL_CLIP_PLANE1);
    glEnable(GL_CLIP_PLANE2);
    glEnable(GL_CLIP_PLANE3);
    drawShapes(1);
    glDisable(GL_CLIP_PLANE0);
    glDisable(GL_CLIP_PLANE1);
    glDisable(GL_CLIP_PLANE2);
    glDisable(GL_CLIP_PLANE3);

    // 3. Clipping window border
    glColor3f(1.0f, 1.0f, 0.0f);
    glLineWidth(2.0f);
    glBegin(GL_LINE_LOOP);
       glVertex2f(-0.5f, -0.5f);
       glVertex2f( 0.5f, -0.5f);
       glVertex2f( 0.5f, 0.5f);
       glVertex2f(-0.5f, 0.5f);
    glEnd();

    glFlush();
}

int main(int argc, char** argv) {
  glutInit(&argc, argv);
  glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
  glutInitWindowSize(500, 500);
  glutCreateWindow("Clipping on Different Shapes");

    glClearColor(0.0, 0.0, 0.0, 1.0);
    glMatrixMode(GL_PROJECTION);
    glLoadIdentity();
    gluOrtho2D(-1, 1, -1, 1);
    glMatrixMode(GL_MODELVIEW);

    glutDisplayFunc(display);
    glutMainLoop();
    return 0;
}

Output

The shapes are drawn in grey without clipping. The parts of the shapes inside the yellow clipping window are redrawn in colour; the parts outside are clipped.
