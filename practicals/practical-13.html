Develop a Mini Project using Various Graphics Functions

Aim

To develop a menu driven mini project (Graphics Studio) using various graphics functions: shapes, animation, 2D transformations, fill patterns and line clipping.

Input

Menu choice (1 to 6): 1 / 2 / 3 / 4 / 5, and 6 to exit

Program / Code

#include <graphics.h>
#include <conio.h>
#include <dos.h>
#include <math.h>
#include <stdio.h>
#include <stdlib.h>

#define INSIDE 0
#define LEFT 1
#define RIGHT 2
#define BOTTOM 4
#define TOP 8

void heading(char *text) {
  cleardevice();
  setcolor(YELLOW);
  settextstyle(DEFAULT_FONT, HORIZ_DIR, 2);
  outtextxy(20, 12, text);
  settextstyle(DEFAULT_FONT, HORIZ_DIR, 1);
  setcolor(WHITE);
  line(0, 40, getmaxx(), 40);
}

void waitKey() {
  setcolor(LIGHTGRAY);
  outtextxy(20, getmaxy() - 20, "Press any key to return to the menu...");
  getch();
}

/* ---------- 3. Transformations ---------- */
void transformations() {
  float x[3] = {60, 140, 100}, y[3] = {160, 160, 100};
  float nx[3], ny[3], rad = 45 * 3.14159 / 180.0;
  float tx = 220, ty = 80, sx = 1.5, sy = 1.5, xp = 60, yp = 160;
  int i, j;

    heading("3. 2D Transformations");
    setcolor(WHITE);
    for (i = 0; i < 3; i++) {
      j = (i + 1) % 3;
      line((int) x[i], (int) y[i], (int) x[j], (int) y[j]);
    }
    outtextxy(60, 175, "Original");

    // Translation
    setcolor(LIGHTRED);
    for (i = 0; i < 3; i++) { nx[i] = x[i] + tx; ny[i] = y[i] + ty; }
    for (i = 0; i < 3; i++) {
       j = (i + 1) % 3;
       line((int) nx[i], (int) ny[i], (int) nx[j], (int) ny[j]);
    }
    outtextxy(270, 255, "Translated");

    // Scaling about the first vertex
    setcolor(LIGHTGREEN);
    for (i = 0; i < 3; i++) {
       nx[i] = xp + (x[i] - xp) * sx;
       ny[i] = yp + (y[i] - yp) * sy;



    }
    for (i = 0; i < 3; i++) {
      j = (i + 1) % 3;
      line((int) nx[i], (int) ny[i], (int) nx[j], (int) ny[j]);
    }
    outtextxy(160, 60, "Scaled");

    // Rotation about the first vertex
    setcolor(LIGHTCYAN);
    for (i = 0; i < 3; i++) {
       nx[i] = xp + (x[i] - xp) * cos(rad) - (y[i] - yp) * sin(rad);
       ny[i] = yp + (x[i] - xp) * sin(rad) + (y[i] - yp) * cos(rad);
    }
    for (i = 0; i < 3; i++) {
       j = (i + 1) % 3;
       line((int) nx[i], (int) ny[i], (int) nx[j], (int) ny[j]);
    }
    outtextxy(60, 260, "Rotated");
    waitKey();
}

/* ---------- 5. Cohen-Sutherland line clipping ---------- */
int xmin = 200, ymin = 120, xmax = 440, ymax = 320;

int computeCode(float x, float y) {
  int code = INSIDE;
  if (x < xmin) code |= LEFT;
  else if (x > xmax) code |= RIGHT;
  if (y < ymin) code |= TOP;
  else if (y > ymax) code |= BOTTOM;
  return code;
}

int clip(float *x1, float *y1, float *x2, float *y2) {
  int code1 = computeCode(*x1, *y1), code2 = computeCode(*x2, *y2), out;
  float x, y;
  while (1) {
     if ((code1 | code2) == 0) return 1;
     if (code1 & code2) return 0;
     out = code1 ? code1 : code2;
     if (out & TOP) {
        x = *x1 + (*x2 - *x1) * (ymin - *y1) / (*y2 - *y1); y = ymin;
     } else if (out & BOTTOM) {
        x = *x1 + (*x2 - *x1) * (ymax - *y1) / (*y2 - *y1); y = ymax;
     } else if (out & RIGHT) {
        y = *y1 + (*y2 - *y1) * (xmax - *x1) / (*x2 - *x1); x = xmax;
     } else {
        y = *y1 + (*y2 - *y1) * (xmin - *x1) / (*x2 - *x1); x = xmin;
     }
     if (out == code1) { *x1 = x; *y1 = y; code1 = computeCode(x, y); }
     else          { *x2 = x; *y2 = y; code2 = computeCode(x, y); }
  }
}

void lineClipping() {
  int ln[4][4] = {{120, 150, 500, 330}, {250, 80, 380, 380},
            {50, 350, 150, 400}, {260, 200, 380, 260}};
  int i;
  float a, b, c, d;

  heading("5. Line Clipping (Cohen-Sutherland)");
  setcolor(CYAN);
  rectangle(xmin, ymin, xmax, ymax);
  for (i = 0; i < 4; i++) {
    setcolor(DARKGRAY);
    setlinestyle(DOTTED_LINE, 0, 1);
    line(ln[i][0], ln[i][1], ln[i][2], ln[i][3]);
    a = ln[i][0]; b = ln[i][1]; c = ln[i][2]; d = ln[i][3];
    if (clip(&a, &b, &c, &d)) {
       setcolor(LIGHTGREEN);
       setlinestyle(SOLID_LINE, 0, THICK_WIDTH);
       line((int) a, (int) b, (int) c, (int) d);
    }
  }



    setlinestyle(SOLID_LINE, 0, 1);
    setcolor(WHITE);
    outtextxy(20, 50, "Dotted: original lines Green: clipped lines");
    waitKey();
}

         case '3': transformations(); break;
         case '4': fillPatterns(); break;
         case '5': lineClipping(); break;
       }
    } while (ch != '6');

    closegraph();
    return 0;
}
