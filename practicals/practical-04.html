Midpoint Circle and Midpoint Ellipse Drawing Algorithms

a) Implement midpoint circle drawing algorithm.b) Implement midpoint ellipse drawing algorithm.

01a) Implement midpoint circle drawing algorithm.

Aim

To implement the Midpoint Circle Drawing algorithm.

Input

Enter centre (xc, yc): 250 200 Enter radius: 100

Program / Code

#include <graphics.h>
#include <conio.h>
#include <stdio.h>

void plotPoints(int xc, int yc, int x, int y) {
  putpixel(xc + x, yc + y, WHITE);
  putpixel(xc - x, yc + y, WHITE);
  putpixel(xc + x, yc - y, WHITE);
  putpixel(xc - x, yc - y, WHITE);
  putpixel(xc + y, yc + x, WHITE);
  putpixel(xc - y, yc + x, WHITE);
  putpixel(xc + y, yc - x, WHITE);
  putpixel(xc - y, yc - x, WHITE);
}

void midpointCircle(int xc, int yc, int r) {
  int x = 0, y = r;
  int p = 1 - r;        // Initial decision parameter

    plotPoints(xc, yc, x, y);
    while (x < y) {
      x++;
      if (p < 0) {
         p += 2 * x + 1;
      } else {
         y--;
         p += 2 * (x - y) + 1;
      }
      plotPoints(xc, yc, x, y);
    }
}

int main() {
  int gd = DETECT, gm, xc, yc, r;

    printf("Enter centre (xc, yc): ");
    scanf("%d%d", &xc, &yc);
    printf("Enter radius: ");
    scanf("%d", &r);



    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
    midpointCircle(xc, yc, r);

    getch();
    closegraph();
    return 0;
}

Output

A circle of radius 100 with centre (250, 200) is drawn in the graphics window.

02b) Implement midpoint ellipse drawing algorithm.

Aim

To implement the Midpoint Ellipse Drawing algorithm.

Input

Enter centre (xc, yc): 250 200 Enter radii (rx, ry): 120 60

Program / Code

#include <graphics.h>
#include <conio.h>
#include <stdio.h>

void plotEllipse(int xc, int yc, int x, int y) {
  putpixel(xc + x, yc + y, WHITE);
  putpixel(xc - x, yc + y, WHITE);
  putpixel(xc + x, yc - y, WHITE);
  putpixel(xc - x, yc - y, WHITE);
}

void midpointEllipse(int xc, int yc, int rx, int ry) {
  float dx, dy, p1, p2;



    float rx2 = (float) rx * rx, ry2 = (float) ry * ry;
    int x = 0, y = ry;

    // Region 1
    p1 = ry2 - rx2 * ry + 0.25 * rx2;
    dx = 2 * ry2 * x;
    dy = 2 * rx2 * y;
    while (dx < dy) {
       plotEllipse(xc, yc, x, y);
       x++;
       dx += 2 * ry2;
       if (p1 < 0) {
          p1 += dx + ry2;
       } else {
          y--;
          dy -= 2 * rx2;
          p1 += dx - dy + ry2;
       }
    }

    // Region 2
    p2 = ry2 * (x + 0.5) * (x + 0.5) + rx2 * (y - 1) * (y - 1) - rx2 * ry2;
    while (y >= 0) {
       plotEllipse(xc, yc, x, y);
       y--;
       dy -= 2 * rx2;
       if (p2 > 0) {
          p2 += rx2 - dy;
       } else {
          x++;
          dx += 2 * ry2;
          p2 += dx - dy + rx2;
       }
    }
}

int main() {
  int gd = DETECT, gm, xc, yc, rx, ry;

    printf("Enter centre (xc, yc): ");
    scanf("%d%d", &xc, &yc);
    printf("Enter radii (rx, ry): ");
    scanf("%d%d", &rx, &ry);

    initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
    midpointEllipse(xc, yc, rx, ry);

    getch();
    closegraph();
    return 0;
}

Output

An ellipse with rx = 120 and ry = 60 and centre (250, 200) is drawn in the graphics window.
