Implementation of 2D Transformations (Translation, Rotation, Scaling, Reflection and Shearing)

Aim

To implement 2D transformations: translation, rotation, scaling, reflection and shearing.

Input

Enter translation (tx, ty): 100 50 Enter rotation angle in degrees: 45 Enter scaling factors (sx, sy): 1.5 1.5 Enter shearing factors (shx, shy): 0.5 0.5

Program / Code

#include <graphics.h>
#include <conio.h>
#include <math.h>
#include <stdio.h>

int cx, cy;               // screen position of the origin

void drawAxes() {
  setcolor(DARKGRAY);
  line(0, cy, getmaxx(), cy);
  line(cx, 0, cx, getmaxy());
}

void drawTriangle(float x[], float y[], int color) {
  int i, j;
  setcolor(color);
  for (i = 0; i < 3; i++) {
    j = (i + 1) % 3;
    line(cx + (int) x[i], cy - (int) y[i], cx + (int) x[j], cy - (int) y[j]);
  }
}

void show(char *title, float x[], float y[], float nx[], float ny[]) {
  cleardevice();
  drawAxes();
  drawTriangle(x, y, WHITE);
  drawTriangle(nx, ny, LIGHTRED);
  setcolor(YELLOW);
  outtextxy(10, 10, title);
  outtextxy(10, 25, "White: Original Red: Transformed");
  getch();
}

int main() {
  int gd = DETECT, gm, i;
  float x[3] = {20, 120, 70};
  float y[3] = {20, 20, 100};



 float nx[3], ny[3];
 float tx, ty, angle, rad, sx, sy, shx, shy;

 printf("Enter translation (tx, ty): ");
 scanf("%f%f", &tx, &ty);
 printf("Enter rotation angle in degrees: ");
 scanf("%f", &angle);
 printf("Enter scaling factors (sx, sy): ");
 scanf("%f%f", &sx, &sy);
 printf("Enter shearing factors (shx, shy): ");
 scanf("%f%f", &shx, &shy);

 initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");
 cx = getmaxx() / 2;
 cy = getmaxy() / 2;

 // 1. Translation
 for (i = 0; i < 3; i++) { nx[i] = x[i] + tx; ny[i] = y[i] + ty; }
 show("Translation", x, y, nx, ny);

 // 2. Rotation about the origin
 rad = angle * 3.14159 / 180.0;
 for (i = 0; i < 3; i++) {
    nx[i] = x[i] * cos(rad) - y[i] * sin(rad);
    ny[i] = x[i] * sin(rad) + y[i] * cos(rad);
 }
 show("Rotation", x, y, nx, ny);

 // 3. Scaling about the origin
 for (i = 0; i < 3; i++) { nx[i] = x[i] * sx; ny[i] = y[i] * sy; }
 show("Scaling", x, y, nx, ny);

 // 4. Reflection about X-axis
 for (i = 0; i < 3; i++) { nx[i] = x[i]; ny[i] = -y[i]; }
 show("Reflection about X-axis", x, y, nx, ny);

 // Reflection about Y-axis
 for (i = 0; i < 3; i++) { nx[i] = -x[i]; ny[i] = y[i]; }
 show("Reflection about Y-axis", x, y, nx, ny);

 // Reflection about the origin
 for (i = 0; i < 3; i++) { nx[i] = -x[i]; ny[i] = -y[i]; }
 show("Reflection about Origin", x, y, nx, ny);

 // 5. Shearing along X-axis
 for (i = 0; i < 3; i++) { nx[i] = x[i] + shx * y[i]; ny[i] = y[i]; }
 show("Shearing along X-axis", x, y, nx, ny);

 // Shearing along Y-axis
 for (i = 0; i < 3; i++) { nx[i] = x[i]; ny[i] = y[i] + shy * x[i]; }
 show("Shearing along Y-axis", x, y, nx, ny);

 closegraph();



    return 0;
}

Output

For every transformation, the original triangle (white) and the transformed triangle (red) are displayed. Press any key to move to the next transformation.
