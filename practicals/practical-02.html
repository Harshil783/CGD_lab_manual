PRACTICAL 02 / 13

Perform Animation such as Rising Sun, Moving Vehicle, Smileys, Screen saver etc.

Aim

To perform animation in C/C++ graphics: a rising sun, a moving vehicle and a bouncing smiley (screen saver).

Input

No user input is required. Press any key to move from one animation to the next.

Program / Code

#include <graphics.h>
#include <conio.h>
#include <dos.h>

void risingSun() {
  int y, w = getmaxx(), h = getmaxy();
  for (y = h - 20; y >= 100; y -= 2) {
    cleardevice();
    // Sun
    setcolor(YELLOW);
    setfillstyle(SOLID_FILL, YELLOW);
    fillellipse(w / 2, y, 40, 40);
    // Ground (drawn after the sun so the sun rises from behind it)
    setcolor(GREEN);
    setfillstyle(SOLID_FILL, GREEN);
    bar(0, h - 60, w, h);
    delay(30);
  }
}

void movingVehicle() {
  int x, w = getmaxx();
  for (x = -140; x < w; x += 4) {
    cleardevice();
    // Road
    setcolor(WHITE);
    line(0, 300, w, 300);
    // Car body
    setcolor(RED);
    setfillstyle(SOLID_FILL, RED);
    bar(x, 240, x + 140, 280);
    bar(x + 30, 205, x + 110, 240);
    // Windows
    setfillstyle(SOLID_FILL, LIGHTCYAN);
    bar(x + 40, 212, x + 65, 238);
    bar(x + 75, 212, x + 100, 238);
    // Wheels
    setcolor(WHITE);
    setfillstyle(SOLID_FILL, DARKGRAY);
    fillellipse(x + 35, 285, 15, 15);
    fillellipse(x + 105, 285, 15, 15);


        delay(20);
        if (kbhit()) {
           getch();
           break;
        }
    }
}

void smiley() {
  int x = 100, y = 100, dx = 5, dy = 4, r = 40;
  int w = getmaxx(), h = getmaxy();
  while (!kbhit()) {
    cleardevice();
    // Face
    setcolor(YELLOW);
    setfillstyle(SOLID_FILL, YELLOW);
    fillellipse(x, y, r, r);
    // Eyes and mouth
    setcolor(BLACK);
    setfillstyle(SOLID_FILL, BLACK);
    fillellipse(x - 15, y - 12, 5, 7);
    fillellipse(x + 15, y - 12, 5, 7);
    arc(x, y + 5, 200, 340, 25);
    // Bounce off the screen edges
    x += dx;
    y += dy;
    if (x - r <= 0 || x + r >= w) dx = -dx;
    if (y - r <= 0 || y + r >= h) dy = -dy;
    delay(25);
  }
  getch();
}

int main() {
  int gd = DETECT, gm;
  initgraph(&gd, &gm, "C:\\TURBOC3\\BGI");

    risingSun();
    outtextxy(200, 20, "Rising Sun - press any key");
    getch();

    movingVehicle();
    outtextxy(200, 20, "Moving Vehicle done - press any key");
    getch();

    smiley(); // screen saver style bouncing smiley (any key stops it)

    closegraph();
    return 0;
}

Output

The sun rises above the ground, then the car moves across the screen, and finally a smiley bounces around the window until a key is pressed.
